function speakPromptSafe(text, callback) {
    let called = false;
    const done = () => {
        if (called) return;
        called = true;
        window.currentUtterance = null;
        if (callback) callback();
    };

    try {
        if (!('speechSynthesis' in window)) {
            done();
            return;
        }
        
        window.speechSynthesis.cancel();
        
        const u = new SpeechSynthesisUtterance(text);
        window.currentUtterance = u; 
        if (systemVoice) u.voice = systemVoice;
        u.onend = done;
        u.onerror = done;
        window.speechSynthesis.speak(u);
        
        // Failsafe timeout guarantees it NEVER freezes
        setTimeout(done, (text.length * 70) + 2000);
    } catch(e) {
        done();
    }
}

function setupSpeechRecognition() {
    try {
        const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
        if (SR) {
            speechRecognition = new SR();
            speechRecognition.continuous = true;
            speechRecognition.interimResults = true;
            speechRecognition.lang = 'en-US';

            speechRecognition.onstart = () => {
                document.getElementById('speechStateBadge').textContent = "● Mic Active";
                document.getElementById('speechStateBadge').style.color = "var(--success)";
            };

            speechRecognition.onresult = (e) => {
                if (isTyping) return;

                let finalStr = "";
                let interimStr = "";
                for (let i = e.resultIndex; i < e.results.length; i++) {
                    if (e.results[i].isFinal) finalStr += e.results[i][0].transcript;
                    else interimStr += e.results[i][0].transcript;
                }

                if (finalStr) {
                    transcriptBuffer += " " + finalStr;
                    document.getElementById('liveTranscriptDisplay').textContent = transcriptBuffer;
                    analyzeFillerWordsInText(finalStr);
                } else if (interimStr) {
                    document.getElementById('liveTranscriptDisplay').textContent = transcriptBuffer + " " + interimStr;
                }
            };

            speechRecognition.onerror = () => {
                document.getElementById('speechStateBadge').textContent = "⌨️ Manual Entry Active";
            };

            speechRecognition.onend = () => {
                if (!isAiSpeaking && !isProcessing && currentQuestionIndex < sessionQuestions.length) {
                    try { speechRecognition.start(); } catch (err) {}
                }
            };
        }
    } catch(e) {}
}

function terminateSessionEarly() {
    if (isProcessing) return;
    
    let fullAnswer = transcriptBuffer.trim();
    const typedText = document.getElementById('userTextInput').value.trim();
    if (typedText) {
        fullAnswer += (fullAnswer ? " " : "") + typedText;
    }

    const isAnswerEmpty = (!fullAnswer || fullAnswer.includes("wait for the AI") || fullAnswer.includes("Listening for your response"));

    if (!isAnswerEmpty) {
        sessionQuestions = sessionQuestions.slice(0, currentQuestionIndex + 1);
        submitCandidateResponse(false);
    } else {
        sessionQuestions = sessionQuestions.slice(0, currentQuestionIndex);
        completeSession();
    }
}

function submitCandidateResponse(isForcedTimeout = false) {
    if (isProcessing) return;
    isProcessing = true;
    clearInterval(timerInterval);
    stopMic();
    if ('speechSynthesis' in window) window.speechSynthesis.cancel();

    let fullAnswer = transcriptBuffer.trim();
    const typedText = document.getElementById('userTextInput').value.trim();
    if (typedText) {
        fullAnswer += (fullAnswer ? " " : "") + typedText;
        document.getElementById('userTextInput').value = "";
        analyzeFillerWordsInText(typedText);
    }

    if (!fullAnswer || fullAnswer.includes("Listening for your response") || fullAnswer.includes("wait for the AI")) {
        fullAnswer = isForcedTimeout ? "Response timed out under pressure timer." : "No response submitted.";
    }

    const currentQuestion = sessionQuestions[currentQuestionIndex];
    
    if (currentQuestion) {
        const keys = currentQuestion.keys || [];
        const lower = fullAnswer.toLowerCase();
        const words = lower.split(/\s+/).filter(w => w.length > 0);

        let hits = 0;
        keys.forEach(k => { if (lower.includes(k.toLowerCase())) hits++; });

        let score = 0;
        if (fullAnswer.includes("timed out") || fullAnswer.includes("No response")) {
            score = 0;
        } else if (hits >= 2) {
            score = 100;
        } else if (hits === 1) {
            score = 85;
        } else {
            score = words.length >= 10 ? 75 : (words.length >= 4 ? 60 : 35);
        }

        totalAccuracyScore += score;
        evaluatedQuestionsCount++;

        userAnswersLog.push({
            q: currentQuestion.q,
            candidateAns: fullAnswer,
            idealAns: currentQuestion.ideal,
            score: score
        });
    }

    currentQuestionIndex++;
    presentQuestion();
}

    else { badge.className = "timer-pill"; bar.style.backgroundColor = "var(--primary)"; }
}

/* =====================================================
   FILLER WORD & FLUENCY TRACKER
===================================================== */
function analyzeFillerWordsInText(text) {
    const lower = text.toLowerCase();
    let detectedCount = 0;
    FILLER_WORDS.forEach(filler => {
        const regex = new RegExp(`\\b${filler}\\b`, 'gi');
        const matches = lower.match(regex);
        if (matches) detectedCount += matches.length;
    });

    sessionFillerWordCount += detectedCount;
    document.getElementById('liveFillerCountDisplay').textContent = `${sessionFillerWordCount} words`;
    const words = lower.split(/\s+/).filter(w => w.length > 0);
    const ratio = words.length > 0 ? (detectedCount / words.length) : 0;
    const fluencyPct = Math.max(50, Math.floor(100 - (ratio * 150)));
    document.getElementById('liveFluencyRatingDisplay').textContent = `${fluencyPct}%`;
}

/* =====================================================
   AUDIO ANALYSIS & VISUALIZER (Web Audio API)
===================================================== */
function startAudioAnalysis(stream) {
    try {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        analyser = audioCtx.createAnalyser();
        analyser.fftSize = 64;
        micSource = audioCtx.createMediaStreamSource(stream);
        micSource.connect(analyser);

        const canvas = document.getElementById('audioVisualizerCanvas');
        const ctx = canvas.getContext('2d');
        const bufferLength = analyser.frequencyBinCount;
        const dataArray = new Uint8Array(bufferLength);

        function drawVisualizer() {
            visualizerAnimFrame = requestAnimationFrame(drawVisualizer);
            analyser.getByteFrequencyData(dataArray);
            ctx.fillStyle = '#000000';
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            let sum = 0;
            const barWidth = (canvas.width / bufferLength) * 1.5;
            let x = 0;

            for (let i = 0; i < bufferLength; i++) {
                const barHeight = (dataArray[i] / 255) * canvas.height;
                sum += dataArray[i];
                ctx.fillStyle = `rgb(79, 70, ${229 + dataArray[i] / 2})`;
                ctx.fillRect(x, canvas.height - barHeight, barWidth, barHeight);
                x += barWidth + 1;
            }

            const avg = sum / bufferLength;
            totalAudioSamples++;
            if (avg < 15) lowVolSamples++;
            else if (avg > 180) loudVolSamples++;

            const ratingText = document.getElementById('volumeRatingText');
            if (avg < 15) { ratingText.textContent = "Level: Soft"; ratingText.style.color = "#facc15"; } 
            else if (avg > 180) { ratingText.textContent = "Level: High"; ratingText.style.color = "#f87171"; } 
            else { ratingText.textContent = "Level: Optimal"; ratingText.style.color = "#34d399"; }
        }
        drawVisualizer();
    } catch (e) {}
}

function stopAudioAnalysis() {
    if (visualizerAnimFrame) cancelAnimationFrame(visualizerAnimFrame);
    if (audioCtx) { audioCtx.close(); audioCtx = null; }
}

/* =====================================================
   SESSION VIDEO & AUDIO RECORDER
===================================================== */
function startMediaRecorder(stream) {
    recordedChunks = [];
    try {
        mediaRecorder = new MediaRecorder(stream, { mimeType: 'video/webm;codecs=vp8,opus' });
    } catch (e) {
        try { mediaRecorder = new MediaRecorder(stream); } catch (err) { return; }
    }
    mediaRecorder.ondataavailable = (e) => { if (e.data.size > 0) recordedChunks.push(e.data); };
    mediaRecorder.onstop = () => {
        const blob = new Blob(recordedChunks, { type: 'video/webm' });
        finalRecordingBlobUrl = URL.createObjectURL(blob);
        document.getElementById('sessionReplayVideo').src = finalRecordingBlobUrl;
    };
    mediaRecorder.start(1000);
}

function stopMediaRecorder() {
    if (mediaRecorder && mediaRecorder.state !== 'inactive') {
        try { mediaRecorder.stop(); } catch (e) {}
    }
}

/* =====================================================
   MEDIAPIPE VISION TELEMETRY
===================================================== */
function initializeVisionTracking() {
    try {
        if (!sharedGlobalStream) return;
        const faceMesh = new FaceMesh({ locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/face_mesh/${file}` });
        faceMesh.setOptions({ maxNumFaces: 1, refineLandmarks: true, minDetectionConfidence: 0.5 });
        faceMesh.onResults(processFaceTelemetry);

        const pose = new Pose({ locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/pose/${file}` });
        pose.setOptions({ modelComplexity: 1, minDetectionConfidence: 0.5 });
        pose.onResults(processPoseTelemetry);

        const videoEl = document.getElementById('activeCamFeed');
        const camera = new Camera(videoEl, {
            onFrame: async () => {
                if (sharedGlobalStream) {
                    await faceMesh.send({ image: videoEl });
                    await pose.send({ image: videoEl });
                }
            },
            width: 320,
            height: 240
        });
        camera.start();
    } catch (e) {
        console.warn("MediaPipe Vision tracking bypassed:", e);
    }
}

function processFaceTelemetry(results) {
    if (results.multiFaceLandmarks && results.multiFaceLandmarks.length > 0) {
        totalFocusSamples++;
        const landmarks = results.multiFaceLandmarks[0];
        const nose = landmarks[1];

        if (nose.x > 0.35 && nose.x < 0.65 && nose.y > 0.35 && nose.y < 0.65) {
            goodFocusSamples++;
            document.getElementById('visionFrameStatus').textContent = "Centered";
            document.getElementById('visionFrameStatus').style.color = "#34d399";
        } else {
            document.getElementById('visionFrameStatus').textContent = "Deviated";
            document.getElementById('visionFrameStatus').style.color = "#f87171";
        }

        if (prevNosePosition) {
            const delta = Math.hypot(nose.x - prevNosePosition.x, nose.y - prevNosePosition.y);
            if (delta > 0.08) {
                excessiveHeadMovements++;
                document.getElementById('visionHeadStatus').textContent = "Nodding / Shifting";
                document.getElementById('visionHeadStatus').style.color = "#facc15";
            } else {
                document.getElementById('visionHeadStatus').textContent = "Stable";
                document.getElementById('visionHeadStatus').style.color = "#34d399";
            }
        }
        prevNosePosition = { x: nose.x, y: nose.y };

        const mouthDist = Math.hypot(landmarks[291].x - landmarks[61].x, landmarks[291].y - landmarks[61].y);
        const faceDist = Math.hypot(landmarks[454].x - landmarks[234].x, landmarks[454].y - landmarks[234].y);
        
        totalExpressionSamples++;
        if ((mouthDist / faceDist) > 0.29) {
            positiveExpressionSamples += 100;
            document.getElementById('visionSentimentStatus').textContent = "Confident / Smiling";
        } else {
            positiveExpressionSamples += 80;
            document.getElementById('visionSentimentStatus').textContent = "Calm / Focused";
        }
    }
}

function processPoseTelemetry(results) {
    if (results.poseLandmarks) {
        totalPostureSamples++;
        const leftWrist = results.poseLandmarks[15];
        const rightWrist = results.poseLandmarks[16];
        const nose = results.poseLandmarks[0];

        if (nose && (leftWrist || rightWrist)) {
            const leftDist = leftWrist ? Math.hypot(leftWrist.x - nose.x, leftWrist.y - nose.y) : 1;
            const rightDist = rightWrist ? Math.hypot(rightWrist.x - nose.x, rightWrist.y - nose.y) : 1;

            if (leftDist < 0.18 || rightDist < 0.18) {
                totalFidgetAlerts++;
                document.getElementById('visionFidgetStatus').textContent = "Hand near Face";
                document.getElementById('visionFidgetStatus').style.color = "#f87171";
            } else {
                document.getElementById('visionFidgetStatus').textContent = "None Detected";
                document.getElementById('visionFidgetStatus').style.color = "#a78bfa";
            }
        }
        if (Math.abs(results.poseLandmarks[11].y - results.poseLandmarks[12].y) < 0.06) goodPostureSamples++;
    }
}

function completeSession() {
    clearInterval(timerInterval);
    stopMic();
    stopAudioAnalysis();
    stopMediaRecorder();
    if ('speechSynthesis' in window) window.speechSynthesis.cancel();

    document.getElementById('interviewDesk').style.display = 'none';
    document.getElementById('reportCard').style.display = 'block';

    const safeCount = Math.max(1, evaluatedQuestionsCount);
    let finalAccuracy = safeCount > 0 ? Math.floor(totalAccuracyScore / safeCount) : 0;
    
    if (userAnswersLog.length === 0) {
        finalAccuracy = 0;
    }

    const finalFocus = totalFocusSamples > 0 ? Math.floor((goodFocusSamples / totalFocusSamples) * 100) : 85;
    const finalPosture = totalPostureSamples > 0 ? Math.floor((goodPostureSamples / totalPostureSamples) * 100) : 90;
    const finalExpression = totalExpressionSamples > 0 ? Math.floor(positiveExpressionSamples / totalExpressionSamples) : 85;

    const fluencyScore = Math.max(50, Math.floor(100 - (sessionFillerWordCount * 4)));
    const fidgetScore = Math.max(50, Math.floor(100 - (totalFidgetAlerts * 3) - (excessiveHeadMovements * 2)));

    let audioToneLabel = "Balanced";
    if (lowVolSamples > (totalAudioSamples * 0.45)) audioToneLabel = "Soft / Monotone";
    else if (loudVolSamples > (totalAudioSamples * 0.25)) audioToneLabel = "High / Expressive";

    const overall = Math.floor((finalAccuracy + finalFocus + finalPosture + finalExpression + fluencyScore + fidgetScore) / 6);

    document.getElementById('overallScoreRing').textContent = `${overall}%`;
    document.getElementById('metricAccuracy').textContent = `${finalAccuracy}%`;
    document.getElementById('metricFluency').textContent = `${fluencyScore}%`;
    document.getElementById('metricFillerCount').textContent = sessionFillerWordCount;
    document.getElementById('metricAudioTone').textContent = audioToneLabel;
    document.getElementById('metricFocus').textContent = `${finalFocus}%`;
    document.getElementById('metricFidgetScore').textContent = `${fidgetScore}%`;
    
    const notesContainer = document.getElementById('coachingNotesFeed');
    notesContainer.innerHTML = "";

    if (sessionFillerWordCount > 3) {
        notesContainer.innerHTML += `
            <div class="coaching-card" style="background:#fffbeb; border-color:#fde68a;">
                <div class="coaching-title" style="color:#b45309;">💬 Verbal Filler Reduction</div>
                <div class="coaching-desc" style="color:#92400e;">Logged ${sessionFillerWordCount} filler phrases. Taking a silent 1-2 second pause demonstrates higher domain control than filling pauses verbally.</div>
            </div>`;
    } else {
        notesContainer.innerHTML += `
            <div class="coaching-card">
                <div class="coaching-title">✅ High Verbal Fluency</div>
                <div class="coaching-desc">Minimal to zero verbal fillers detected. Your speech flow was structured, direct, and concise.</div>
            </div>`;
    }

    if (totalFidgetAlerts > 4 || excessiveHeadMovements > 10) {
        notesContainer.innerHTML += `
            <div class="coaching-card" style="background:#fffbeb; border-color:#fde68a;">
                <div class="coaching-title" style="color:#b45309;">🧍 Physical Presence & Telemetry</div>
                <div class="coaching-desc" style="color:#92400e;">Frequent hand-to-face contact or head shifting recorded. Resting hands flat on the desk or using deliberate open gestures projects steady poise.</div>
            </div>`;
    }

    const reviewContainer = document.getElementById('answerReviewFeed');
    reviewContainer.innerHTML = "";
    
    if (userAnswersLog.length === 0) {
        reviewContainer.innerHTML = `<div class="q-review-card"><div class="review-prompt">No questions were fully answered before ending the session.</div></div>`;
    } else {
        userAnswersLog.forEach((item, index) => {
            reviewContainer.innerHTML += `
                <div class="q-review-card">
                    <div class="review-prompt">Question ${index + 1}: ${item.q}</div>
                    <div class="review-user-ans"><strong>Your Response:</strong> "${item.candidateAns}"</div>
                    <div class="review-ideal-ans"><strong>${selectedTrack} Benchmark Answer:</strong> ${item.idealAns}</div>
                </div>`;
        });
    }

    speakPromptSafe("Diagnostic compiled. Session recording and performance analytics are ready for review.", null);
}
